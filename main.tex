\section{Systémový katalog a obecná pravidla}
\begin{itemize}
    \item \textbf{Názvy objektů v katalogu:} Systémové pohledy (\texttt{USER\_TABLES}, \texttt{USER\_TAB\_COLUMNS}) ukládají názvy objektů \textbf{VELKÝMI PÍSMENY}. Při vyhledávání je nutné použít \texttt{'NAZEV\_TABULKY'} nebo funkci \texttt{UPPER()}.
    \item \textbf{Textové řetězce vs. Názvy objektů:}
    \begin{itemize}
        \item \texttt{' '} (Jednoduché apostrofy): Textové řetězce (např. \texttt{v\_text := 'REPORT';}).
        \item \texttt{" "} (Dvojité uvozovky): Vynucení přesného názvu objektu. Chybné použití na text vyvolá výjimku \texttt{PLS-00201}.
    \end{itemize}
    \item \textbf{Chyba ORA-01652 (Nelze rozšířit temp segment):} Vyčerpání datového limitu v tabulkovém prostoru. Řešením je spuštění \texttt{PURGE RECYCLEBIN;} nebo odstranění nepotřebných tabulek příkazem \texttt{DROP TABLE nazev PURGE;}.
    \item \textbf{Operátor LIKE:} Vyžaduje zástupný znak \texttt{\%} pro shodu vzoru. Bez něj funguje jako standardní operátor rovnosti (např. \texttt{LIKE 'REPORT\%'}).
\end{itemize}

\section{SQL Specifika a Časté Chyby}
\begin{itemize}
    \item \textbf{CTAS (Create Table As Select):} Pokud je nová tabulka vytvářena z agregačních funkcí (např. \texttt{COUNT}), je \textbf{nezbytné} definovat alias sloupce (\texttt{COUNT(aid) AS pocet}). Bez aliasu vzniká chyba \texttt{ORA-00998}.
    \item \textbf{Filtrování u LEFT JOIN:} Umístění omezující podmínky pro připojovanou tabulku do klauzule \texttt{WHERE} (např. \texttt{WHERE j.lang = 'NE'}) převede spojení na \texttt{INNER JOIN}. Pro zachování vnějšího spojení musí být podmínka součástí klauzule \texttt{ON}.
    \item \textbf{Nejednoznačný sloupec (ORA-00918):} Při spojování tabulek se stejnojmennými sloupci je nutné specifikovat alias tabulky jak v klauzuli \texttt{SELECT}, tak v klauzuli \texttt{GROUP BY} (např. \texttt{i.iid}).
    \item \textbf{Agregace a SYSTIMESTAMP:} Systémové funkce nepatří do klauzule \texttt{GROUP BY}. Jejich přítomnost rozbije logiku seskupování.
\end{itemize}

\section{Dynamické SQL a Vázané Proměnné}
\begin{itemize}
    \item \textbf{Použití vázaných proměnných (\texttt{USING}):} Lze aplikovat \textbf{výhradně} v příkazech DML (hodnoty v INSERT, UPDATE, DELETE nebo WHERE klauzuli). 
    \item \textbf{Omezení pro DDL:} V DDL příkazech (\texttt{CREATE TABLE}, \texttt{DROP TABLE}, atd.) je použití vázaných proměnných \textbf{zakázáno}. Názvy objektů se musí dynamicky skládat pomocí operátoru zřetězení (\texttt{||}).
    \item \textbf{Klíčová slova (ORA-01745):} Název vázané proměnné nesmí kolidovat s rezervovaným slovem SQL (např. nelze použít \texttt{:year}, vhodné je \texttt{:rok}).
\end{itemize}

\section{Datové modelování a Oracle Academy}
\begin{itemize}
    \item \textbf{Složený primární klíč:} Ve vazební tabulce (M:N) musí být vytvořen složený klíč: \texttt{ALTER TABLE vazba ADD CONSTRAINT pk\_vazba PRIMARY KEY (sloupec\_a, sloupec\_b);}.
    \item \textbf{Vztah 1:1 (Optional to Mandatory):} Logika Oracle Academy striktně umisťuje cizí klíč (FK) \textbf{vždy na povinnou stranu} vztahu (do tabulky s mandatory vztahem).
\end{itemize}

\section{Výjimky (Exceptions)}

\noindent
\begin{minipage}{\textwidth}
\subsection*{Deklarace vlastní výjimky}
Pro situace, které databáze nepovažuje za chybu, ale porušují aplikační logiku.
\begin{lstlisting}[language=SQL]
DECLARE
    prilis_mnoho_zaznamu EXCEPTION;
    v_pocet INT;
BEGIN
    SELECT COUNT(*) INTO v_pocet FROM z_article;
    IF v_pocet > 20 THEN
        RAISE prilis_mnoho_zaznamu;
    ELSE
        INSERT INTO z_article (aid, year) VALUES (99, 2026);
    END IF;
EXCEPTION
    WHEN prilis_mnoho_zaznamu THEN
        dbms_output.put_line('Chyba: Prekrocen limit poctu clanku.');
    WHEN OTHERS THEN
        dbms_output.put_line(SQLERRM);
END;
\end{lstlisting}
\end{minipage}

\section{Přesná Syntaxe a Šablony PL/SQL}

\noindent
\begin{minipage}{\textwidth}
\subsection{1. PROCEDURE (Procedura)}
Procedura nevrací hodnotu napřímo, ale vykonává akce. Lokální proměnné se definují mezi klíčovými slovy \texttt{IS}/\texttt{AS} a \texttt{BEGIN} (nepoužívá se slovo \texttt{DECLARE}).
\textbf{Šablona:}
\begin{lstlisting}[language=SQL]
CREATE [OR REPLACE] PROCEDURE jmeno_procedury 
    [(jmeno_parametru [IN | OUT | IN OUT] datovy_typ, ...)] 
IS | AS
    -- definice lokálních proměnných
BEGIN
    -- tělo procedury
[EXCEPTION
    -- zpracování chyb]
END [jmeno_procedury];
\end{lstlisting}

\textbf{Příklad:}
\begin{lstlisting}[language=SQL]
CREATE OR REPLACE PROCEDURE p_vypis_autora(p_rid INT) 
IS
    v_jmeno z_author.name%TYPE;
BEGIN
    SELECT name INTO v_jmeno FROM z_author WHERE rid = p_rid;
    dbms_output.put_line('Autor: ' || v_jmeno);
EXCEPTION
    WHEN NO_DATA_FOUND THEN
        dbms_output.put_line('Autor s ID ' || p_rid || ' neexistuje.');
END p_vypis_autora;
\end{lstlisting}
\end{minipage}

\vspace{2em}
\noindent
\begin{minipage}{\textwidth}
\subsection{2. FUNCTION (Funkce)}
Funkce musí vždy obsahovat návratový typ v definici a musí vracet hodnotu pomocí příkazu \texttt{RETURN}. \texttt{RETURN} musí pokrývat i blok \texttt{EXCEPTION}.
\textbf{Šablona:}
\begin{lstlisting}[language=SQL]
CREATE [OR REPLACE] FUNCTION jmeno_funkce 
    [(jmeno_parametru [IN] datovy_typ, ...)] 
    RETURN navratovy_typ
IS | AS
    -- definice lokálních proměnných
BEGIN
    -- tělo funkce
    RETURN hodnota;
[EXCEPTION
    -- zpracování chyb (včetně RETURN)]
END [jmeno_funkce];
\end{lstlisting}

\textbf{Příklad:}
\begin{lstlisting}[language=SQL]
CREATE OR REPLACE FUNCTION f_pocet_clanku(p_rid INT) 
    RETURN INT 
IS
    v_pocet INT := 0;
BEGIN
    SELECT COUNT(aid) INTO v_pocet 
    FROM z_article_author 
    WHERE rid = p_rid;
    
    RETURN v_pocet;
EXCEPTION
    WHEN OTHERS THEN
        RETURN -1; -- Ošetření případného selhání
END f_pocet_clanku;
\end{lstlisting}
\end{minipage}

\noindent
\begin{minipage}{\textwidth}
\subsection{3. TRIGGER (Spoušť)}
Trigger je automaticky vyvolán při specifikované DML události. Upravovat zapisovaná data přes \texttt{:NEW} lze \textbf{pouze} v \texttt{BEFORE} triggeru.
\textbf{Šablona:}
\begin{lstlisting}[language=SQL]
CREATE [OR REPLACE] TRIGGER jmeno_triggeru
    {BEFORE | AFTER | INSTEAD OF} 
    {INSERT [OR] UPDATE [OR] DELETE} [OF jmeno_sloupce]
    ON jmeno_tabulky
    [REFERENCING OLD AS stara_hodnota NEW AS nova_hodnota]
    [FOR EACH ROW [WHEN (podminka)]]
[DECLARE
    -- definice lokálních proměnných]
BEGIN
    -- tělo triggeru
END [jmeno_triggeru];
\end{lstlisting}

\textbf{Příklad:}
\begin{lstlisting}[language=SQL]
CREATE OR REPLACE TRIGGER t_institution_count
    BEFORE INSERT OR UPDATE ON INSTITUTION_COUNT
    FOR EACH ROW
DECLARE
    v_pocet INT := 0; 
BEGIN
    SELECT COUNT(jou.jid) INTO v_pocet
    FROM Z_INSTITUTION i 
    LEFT JOIN Z_ARTICLE_INSTITUTION ai ON (i.iid = ai.iid)
    LEFT JOIN Z_ARTICLE ar ON (ar.aid = ai.aid)
    LEFT JOIN z_JOURNAL jou ON (jou.jid = ar.jid AND jou.czech_or_slovak = 'NE')
    WHERE i.iid = :NEW.iid;
    
    IF v_pocet = 0 THEN 
        raise_application_error(-20001, 'No article for institution');
    ELSE 
        :NEW.article_count := v_pocet;
        :NEW.stamp := SYSTIMESTAMP;
    END IF;
END t_institution_count;
\end{lstlisting}
\end{minipage}

\noindent
\begin{minipage}{\textwidth}
\subsection{4. KURZORY A CYKLY (FOR IN)}
Implicitní kurzor automaticky zajišťuje otevření dotazu, postupné načítání řádků (\texttt{FETCH}), testování konce a uzavření kurzoru.
\textbf{Šablona a příklad:}
\begin{lstlisting}[language=SQL]
BEGIN
    -- Smycka FOR nepotrebuje DECLARE, OPEN, FETCH ani CLOSE
    FOR zaznam IN (
        SELECT DISTINCT a.year 
        FROM z_Article_institution ai 
        JOIN z_Article a ON (a.aid = ai.aid)
        WHERE ai.iid = 75
        ORDER BY a.year
    ) LOOP
        -- Pristup k datum vyhradne pres teckovou notaci iteracni promenne
        dbms_output.put_line('Aktivni rok: ' || zaznam.year);
    END LOOP;
END;
\end{lstlisting}
\end{minipage}
